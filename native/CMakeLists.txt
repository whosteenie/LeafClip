cmake_minimum_required(VERSION 3.20)
project(LeafClipNative LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The renderer currently requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(clip_bytecode STATIC ShaderPatch.cpp Geometry.cpp third_party/DXBCChecksum.cpp)
target_compile_definitions(clip_bytecode PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(clip_bytecode PUBLIC d3d11 d3dcompiler dxguid)
target_include_directories(clip_bytecode PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(LeafClipNative SHARED Native.cpp D3D12Slots.cpp D3D12Stream.cpp D3D12Root.cpp D3D12PipelineCopy.cpp D3D12RootArguments.cpp D3D12DescriptorCopy.cpp D3D12Resources.cpp)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
  third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
set_target_properties(LeafClipNative PROPERTIES OUTPUT_NAME "LeafClip.Native")
target_link_libraries(LeafClipNative PRIVATE clip_bytecode minhook d3d12)
add_executable(renderer_tests tests/RendererTests.cpp)
target_link_libraries(renderer_tests PRIVATE clip_bytecode LeafClipNative)
add_executable(d3d12_pipeline_tests tests/D3D12PipelineTests.cpp D3D12Stream.cpp D3D12Root.cpp D3D12PipelineCopy.cpp D3D12RootArguments.cpp D3D12DescriptorCopy.cpp D3D12Resources.cpp)
target_link_libraries(d3d12_pipeline_tests PRIVATE clip_bytecode LeafClipNative d3d12 dxgi)
enable_testing()
add_test(NAME renderer_tests COMMAND renderer_tests)
add_test(NAME d3d12_pipeline_tests COMMAND d3d12_pipeline_tests)
